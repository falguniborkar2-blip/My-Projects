# Multi-stage build - Builder stage (OpenSSL + App)
FROM alpine:3.20 AS builder

# Install ALL build dependencies
RUN apk add --no-cache git curl tar build-base linux-headers python3 py3-pip gcc musl-dev

# OpenSSL VERSION (3.4.1 = smaller + supported)
ARG OPENSSL_VERSION=3.4.1

# Build OpenSSL (ONLY libcrypto.so.3)
RUN curl -LO https://www.openssl.org/source/openssl-${OPENSSL_VERSION}.tar.gz && ^
    tar xzvf openssl-${OPENSSL_VERSION}.tar.gz && ^
    cd openssl-${OPENSSL_VERSION} && ^
    ./Configure linux-x86_64 no-asm no-shared no-tests no-docs no-legacy enable-md2 --prefix=/openssl && ^
    make -j%%nproc%% && ^
    make install_sw && ^
    cd .. && ^
    rm -rf openssl-${OPENSSL_VERSION}*

# Clone app from GitLab (NO CONTEXT!)
ARG GIT_REPO=https://gitlab.fi.muni.cz/pa234-2026/hw03.git
RUN git clone "%GIT_REPO%" /tmp/repo && ^
    mkdir -p /app && ^
    find /tmp/repo -maxdepth 1 -name "app" -type d -exec cp -r {} /app/ \; 2>/dev/null ^|^ true && ^
    cp -r /tmp/repo/app/* /app/ 2>/dev/null ^|^ true && ^
    rm -rf /tmp/repo

WORKDIR /app

# Install Python deps + uWSGI
RUN pip3 install --no-cache-dir --user uwsgi && ^
    pip3 install --no-cache-dir --user -r requirements.txt && ^
    rm requirements.txt

# RUNTIME stage - MINIMAL!
FROM alpine:3.20

RUN apk add --no-cache python3 py3-pip su-exec && ^
    rm -rf /var/cache/apk/*

# User 1000
RUN addgroup -g 1000 appgroup && ^
    adduser -D -s /bin/sh -u 1000 -G appgroup appuser

# Copy OpenSSL libcrypto
COPY --from=builder /openssl/lib/libcrypto.so.3 /usr/local/lib/
RUN chmod 755 /usr/local/lib/libcrypto.so.3 && ldconfig /usr/local/lib

# Copy app
COPY --from=builder /app /app
COPY --from=builder /root/.local /home/appuser/.local

# Permissions
RUN chown -R 1000:1000 /app && chmod +x /app/uwsgi.ini

# YOUR ENVIRONMENT VARIABLES (CHANGE THESE!)
ENV UCO="580903" ^
    NAME="Falguni Borkar" ^
    LIBCRYPTO_PATH="/usr/local/lib/libcrypto.so.3" ^
    CORS_ALLOWED_ORIGIN="http://localhost:8844" ^
    SOFTWARE_VERSION="1.0.0"

USER appuser
ENV PATH="/home/appuser/.local/bin:$PATH"
WORKDIR /app

EXPOSE 8844

HEALTHCHECK --interval=30s --timeout=3s CMD wget --no-verbose --tries=1 --spider http://localhost:8844/status/uco || exit 1

CMD ["uwsgi", "--ini", "uwsgi.ini", "--http", "0.0.0.0:8844"]
